```python
b = "Hello, World!"
print(b[2:5])
# output: llo
```


```python
b = "Hello, World!"
print(b[:5])
# output: hello
```

```python
b = "Hello, World!"  
print(b[2:])

# output: llo,world!
```


```python
b = "Hello, World!"
print(b[-5:-2])

# output: orl
```
